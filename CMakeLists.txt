cmake_minimum_required(VERSION 3.21)
project(Sifa VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIFA_BUILD_TESTS "Build the engine and integration tests" ON)
if(MSVC)
  # Keep the portable application independent of a separate VC++ runtime install.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(sifa_core src/vm.cpp src/settings.cpp src/timing.cpp)
target_include_directories(sifa_core PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(sifa_core PUBLIC Threads::Threads)
if(WIN32)
  target_sources(sifa_core PRIVATE src/hypervisor_windows.cpp)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(sifa_core PRIVATE src/hypervisor_linux.cpp)
else()
  message(FATAL_ERROR "Sifa currently targets Windows x64 and Linux x64")
endif()

if(MSVC)
  target_compile_options(sifa_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sifa_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Embed the UI at configure time: the installed executable needs no web folder,
# Python runtime, Node runtime or network connection.
function(sifa_embed_asset path variable)
  # MSVC limits individual string literals. A byte array preserves UTF-8 exactly
  # without splitting multibyte characters or requiring a build-time interpreter.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${path}" asset_hex HEX)
  string(REGEX REPLACE "(..)" "char(0x\\1),\n" asset_bytes "${asset_hex}")
  set(${variable} "${asset_bytes}" PARENT_SCOPE)
endfunction()
sifa_embed_asset(web/index.html SIFA_HTML_BYTES)
sifa_embed_asset(web/styles.css SIFA_CSS_BYTES)
sifa_embed_asset(web/app.js SIFA_JS_BYTES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  web/index.html web/styles.css web/app.js)
configure_file(src/assets.hpp.in generated/assets.hpp @ONLY)
add_executable(sifa src/main.cpp src/server.cpp)
target_include_directories(sifa PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(sifa PRIVATE sifa_core)
if(WIN32)
  target_compile_definitions(sifa_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(sifa PRIVATE ws2_32 bcrypt shell32)
  set_target_properties(sifa PROPERTIES OUTPUT_NAME "Sifa")
endif()
if(MSVC)
  target_compile_options(sifa PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(sifa PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(sifa PRIVATE -static -static-libgcc -static-libstdc++)
endif()

include(CTest)
if(BUILD_TESTING AND SIFA_BUILD_TESTS)
  foreach(suite IN ITEMS vm settings timing)
    add_executable(${suite}_tests tests/${suite}_tests.cpp)
    target_link_libraries(${suite}_tests PRIVATE sifa_core)
    add_test(NAME ${suite} COMMAND ${suite}_tests)
  endforeach()
  if(NOT CMAKE_CROSSCOMPILING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
      add_test(NAME api COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/api_tests.py" "$<TARGET_FILE:sifa>")
      set_tests_properties(api PROPERTIES TIMEOUT 60)
    endif()
  endif()
endif()

install(TARGETS sifa RUNTIME DESTINATION bin)
